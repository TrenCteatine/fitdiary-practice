3import json

8class JsonFormatter(logging.Formatter): 

9 """Формат без содержимого дневника и параметров профиля.""" 

10 

11 def format(self, record): 

12 data = {"time": self.formatTime(record), "level": record.levelname, "logger": record.name, "message": record.getMessage()} 

13 if record.exc_info: 

14 data["exception"] = self.formatException(record.exc_info) 

15 return json.dumps(data, ensure_ascii=False)

18def configure_logging(directory): 

19 """Открыть журнал 1 МБ, две резервные копии.""" 

20 directory.mkdir(parents=True, exist_ok=True) 

21 handler = RotatingFileHandler(directory / "fitdiary.jsonl", maxBytes=1_000_000, backupCount=2, encoding="utf-8") 

22 handler.setFormatter(JsonFormatter()) 

23 logger = logging.getLogger("fitdiary") 

24 logger.setLevel(logging.INFO) 

25 for previous in logger.handlers[:]: 

26 previous.close() 

27 logger.removeHandler(previous) 

28 logger.addHandler(handler)
