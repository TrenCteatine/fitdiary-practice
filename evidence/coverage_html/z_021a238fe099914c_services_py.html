54class DailyTotals: 

55 """Однопроходная агрегация O(n + d log d), d — число активных дней.""" 

56 

57 def calculate(self, state, start, end): 

58 """Посчитать КБЖУ и тренировки включительно между двумя датами.""" 

59 start, end = day(start), day(end) 

60 if start > end: 

61 raise ValidationError("Начало периода позже конца") 

62 totals = defaultdict(lambda: [0.0] * 6) 

63 for meal in state.meals: 

64 if start <= meal.date <= end: 

65 row = totals[meal.date] 

66 for i, value in enumerate(meal.nutrients): 

67 row[i] += value 

68 for workout in state.workouts: 

69 if start <= workout.date <= end: 

70 totals[workout.date][4] += workout.minutes 

71 totals[workout.date][5] += workout.kcal 

72 return [dict(zip(("date", "kcal", "protein", "fat", "carbs", "minutes", "burned"), (key, *totals[key]), strict=True)) for key in sorted(totals)]

108 def remove(self, entry_id): 

109 """Удалить запись по ID; неизвестный ID является ошибкой.""" 

110 meals = [m for m in self.state.meals if m.id != entry_id] 

111 workouts = [w for w in self.state.workouts if w.id != entry_id] 

112 if len(meals) == len(self.state.meals) and len(workouts) == len(self.state.workouts): 

113 raise ValidationError("Запись не найдена") 

114 self._commit(replace(self.state, meals=meals, workouts=workouts))
