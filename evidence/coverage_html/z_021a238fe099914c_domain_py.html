17def number(value, label, minimum=0, maximum=100000): 

18 """Проверить конечное число, принимая запятую как десятичный знак.""" 

19 if isinstance(value, bool): 

20 raise ValidationError(f"{label}: требуется число") 

21 try: 

22 result = float(str(value).replace(",", ".")) 

23 except (ValueError, TypeError): 

24 raise ValidationError(f"{label}: требуется число") from None 

25 if not math.isfinite(result) or not minimum <= result <= maximum: 

26 raise ValidationError(f"{label}: допустимо от {minimum:g} до {maximum:g}") 

27 return result

30def text(value, label, maximum=120): 

31 """Нормализовать обязательную текстовую строку.""" 

32 if not isinstance(value, str) or not value.strip() or len(value.strip()) > maximum: 

33 raise ValidationError(f"{label}: от 1 до {maximum} символов") 

34 return value.strip()

37def day(value): 

38 """Проверить и вернуть дату в формате ГГГГ-ММ-ДД.""" 

39 try: 

40 return date.fromisoformat(value).isoformat() 

41 except (ValueError, TypeError): 

42 raise ValidationError("Дата: используйте ГГГГ-ММ-ДД") from None

45@dataclass(frozen=True) 

46class Profile: 

47 """Параметры пользователя; цель калорий задается самим пользователем.""" 

48 

49 name: str = "Пользователь" 

50 weight: float = 70.0 

51 height: float = 175.0 

52 calorie_goal: float = 2000.0 

53 

54 def __post_init__(self): 

55 object.__setattr__(self, "name", text(self.name, "Имя")) 

56 for key, lo, hi in [("weight", 1, 600), ("height", 30, 300), ("calorie_goal", 1, 20000)]: 

57 object.__setattr__(self, key, number(getattr(self, key), key, lo, hi))

59 @property 

60 def bmi(self): 

61 """ИМТ = масса в кг / квадрат роста в метрах, без медицинской оценки.""" 

62 return self.weight / (self.height / 100) ** 2
